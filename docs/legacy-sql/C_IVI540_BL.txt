get_bihrs(a.part_no) bi_hrs from
 IVM05 a,
 ( SELECT a.mo_no,qty2,qty,A.DIVISION_CODE FROM
    (  SELECT mo_no,SUM(nvl(qty1,0)+nvl(qty2,0)) qty2,DIVISION_CODE FROM
       (   SELECT J.* FROM apitech.IVM540 J,
          ( SELECT PCB,PROD_DATE,MAX(VERSION) VERSION,FAC_CODE FROM APITECH.IVM540
             WHERE prod_date BETWEEN :ED_ST AND :ED_END
            GROUP BY PCB,PROD_DATE,FAC_CODE ) K
          WHERE J.PCB=K.PCB
            AND J.PROD_DATE=K.PROD_DATE
            AND J.VERSION=K.VERSION
            AND J.FAC_CODE=K.FAC_CODE
           AND  J. prod_date BETWEEN :ED_ST AND :ED_END
            AND J.DIVISION_CODE=:DIV
           and  (nvl(qty1,0)+nvl(qty2,0))>0  )
      group by mo_no,DIVISION_CODE) a,
    ( select  mo_no,SUM(QTY1+QTY2) qty  FROM
     (  SELECT J.* FROM apitech.IVM540 J,
          ( SELECT PCB,PROD_DATE,MAX(VERSION) VERSION,FAC_CODE FROM APITECH.IVM540
             WHERE prod_date BETWEEN :ED_ST AND :ED_END
            GROUP BY PCB,PROD_DATE,FAC_CODE ) K
          WHERE J.PCB=K.PCB
            AND J.PROD_DATE=K.PROD_DATE
            AND J.VERSION=K.VERSION
            AND J.FAC_CODE=K.FAC_CODE
            AND J.DIVISION_CODE=:DIV
            AND J.prod_date<= :ED_END )
       GROUP BY MO_NO     ) b
      WHERE A.MO_NO=B.MO_NO  ) B,

 (SELECT MO_NO,DIVISION_CODE,
       (CASE WHEN apitech.GET_MO_qty(MO_NO)=SUM(QTY1+QTY2) THEN MAX(PROD_DATE)
        ELSE '' END) PROD_DATE
  FROM  (  SELECT J.* FROM apitech.IVM540 J,
          ( SELECT PCB,PROD_DATE,MAX(VERSION) VERSION,FAC_CODE FROM APITECH.IVM540
             WHERE prod_date BETWEEN :ED_ST AND :ED_END
            GROUP BY PCB,PROD_DATE,FAC_CODE ) K
          WHERE J.PCB=K.PCB
            AND J.PROD_DATE=K.PROD_DATE
            AND J.VERSION=K.VERSION
            AND J.FAC_CODE=K.FAC_CODE
            AND J.DIVISION_CODE=:DIV
            AND J.prod_date<= :ED_END )
   GROUP BY MO_NO,DIVISION_CODE) C
 where a.mo_no=b.mo_no
 AND A.MO_NO=C.MO_NO
